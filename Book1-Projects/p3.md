```python
import os
from dataclasses import dataclass
from dotenv import load_dotenv

load_dotenv()

    latitude: float = float(os.getenv("LATITUDE", "0"))
    longitude: float = float(os.getenv("LONGITUDE", "0"))
    timezone: str = os.getenv("TIMEZONE", "UTC")

def get_settings() -> Settings:
    return Settings()
```

## 4) db.py (SQLite database helper)

```python
import os
import sqlite3
from typing import Dict, Any, Optional

SCHEMA_SQL = """
CREATE TABLE IF NOT EXISTS weather_observations (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    observed_at TEXT NOT NULL,
    latitude REAL NOT NULL,
    longitude REAL NOT NULL,
    temperature_c REAL NOT NULL,
    humidity_percent REAL NOT NULL,
    wind_kph REAL NOT NULL,
    rain_mm REAL NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_weather_observed_at
ON weather_observations(observed_at);
"""

def ensure_parent_dir(path: str) -> None:
    parent = os.path.dirname(path)
    if parent:
        os.makedirs(parent, exist_ok=True)

def get_conn(db_path: str) -> sqlite3.Connection:
    ensure_parent_dir(db_path)
    conn = sqlite3.connect(db_path)
    conn.execute("PRAGMA journal_mode=WAL;")
    return conn

def init_db(db_path: str) -> None:
    with get_conn(db_path) as conn:
        conn.executescript(SCHEMA_SQL)

def insert_observation(db_path: str, row: Dict[str, Any]) -> None:
    """
    row keys:
      observed_at, latitude, longitude, temperature_c, humidity_percent, wind_kph, rain_mm
    """
    sql = """
    INSERT INTO weather_observations (
        observed_at, latitude, longitude, temperature_c,
        humidity_percent, wind_kph, rain_mm
    )
    VALUES (?, ?, ?, ?, ?, ?, ?);
    """
    vals = (
        row["observed_at"],
        row["latitude"],
        row["longitude"],
        row["temperature_c"],
        row["humidity_percent"],
        row["wind_kph"],
        row["rain_mm"],
    )
    with get_conn(db_path) as conn:
        conn.execute(sql, vals)

def fetch_day_stats(db_path: str, day_yyyy_mm_dd: str) -> Optional[dict]:
    """
    Returns summary for a day using SQLite aggregation.
    day_yyyy_mm_dd example: "2026-01-08"
    observed_at is ISO like "2026-01-08T17:10:05"
    """
    start = f"{day_yyyy_mm_dd}T00:00:00"
    end = f"{day_yyyy_mm_dd}T23:59:59"

    sql = """
    SELECT
        COUNT(*) as n,
        MIN(temperature_c) as temp_min,
        MAX(temperature_c) as temp_max,
        AVG(temperature_c) as temp_avg,
        MIN(humidity_percent) as hum_min,
        MAX(humidity_percent) as hum_max,
        AVG(humidity_percent) as hum_avg,
        MAX(wind_kph) as wind_max,
        SUM(rain_mm) as rain_total
    FROM weather_observations
    WHERE observed_at BETWEEN ? AND ?;
    """

    with get_conn(db_path) as conn:
        row = conn.execute(sql, (start, end)).fetchone()

    if not row or row[0] == 0:
        return None

    return {
        "count": int(row[0]),
        "temp_min": float(row[1]),
        "temp_max": float(row[2]),
        "temp_avg": float(row[3]),
        "hum_min": float(row[4]),
        "hum_max": float(row[5]),
        "hum_avg": float(row[6]),
        "wind_max": float(row[7]),
        "rain_total": float(row[8]) if row[8] is not None else 0.0,
    }

def fetch_latest_observation(db_path: str) -> Optional[dict]:
    sql = """
    SELECT observed_at, temperature_c, humidity_percent, wind_kph, rain_mm
    FROM weather_observations
    ORDER BY observed_at DESC
    LIMIT 1;
    """
    with get_conn(db_path) as conn:
        row = conn.execute(sql).fetchone()

    if not row:
        return None

    return {
        "observed_at": row[0],
        "temperature_c": float(row[1]),
        "humidity_percent": float(row[2]),
        "wind_kph": float(row[3]),
        "rain_mm": float(row[4]),
    }
```

## 5) providers/weather_openmeteo.py (Data source)

```python
import requests

def get_current_weather(latitude: float, longitude: float, timezone: str) -> dict:
    """
    Uses Open-Meteo (free, no API key).
    Returns:
      observed_at (ISO), temperature_c, humidity_percent, wind_kph, rain_mm
    """
    url = (
        "https://api.open-meteo.com/v1/forecast"
        f"?latitude={latitude}&longitude={longitude}"
        "&current=temperature_2m,relative_humidity_2m,wind_speed_10m,precipitation"
        f"&timezone={timezone}"
    )

    r = requests.get(url, timeout=20)
    r.raise_for_status()
    data = r.json()
    cur = data.get("current", {})

    return {
        "observed_at": str(cur.get("time")),
        "temperature_c": float(cur.get("temperature_2m", 0.0)),
        "humidity_percent": float(cur.get("relative_humidity_2m", 0.0)),
        "wind_kph": float(cur.get("wind_speed_10m", 0.0)),
        "rain_mm": float(cur.get("precipitation", 0.0)),
    }
```

## 6) collect.py (Collect + Store)

```python
from config import Settings
from db import init_db, insert_observation
from providers.weather_openmeteo import get_current_weather

def run_collection(settings: Settings) -> dict:
    init_db(settings.db_path)

    weather = get_current_weather(
        latitude=settings.latitude,
        longitude=settings.longitude,
        timezone=settings.timezone,
    )

    row = {
        "observed_at": weather["observed_at"],
        "latitude": settings.latitude,
        "longitude": settings.longitude,
        "temperature_c": weather["temperature_c"],
        "humidity_percent": weather["humidity_percent"],
        "wind_kph": weather["wind_kph"],
        "rain_mm": weather["rain_mm"],
    }

    insert_observation(settings.db_path, row)
    return row
```

## 7) report.py (Generate daily report)

```python
import os
from datetime import datetime, timedelta
from config import Settings
from db import fetch_day_stats, fetch_latest_observation

def ensure_dir(path: str) -> None:
    os.makedirs(path, exist_ok=True)

def format_num(x: float, nd: int = 1) -> str:
    return f"{x:.{nd}f}"

def generate_daily_report(settings: Settings, day_yyyy_mm_dd: str) -> str:
    ensure_dir(settings.reports_dir)

    stats = fetch_day_stats(settings.db_path, day_yyyy_mm_dd)
    latest = fetch_latest_observation(settings.db_path)

    filename = os.path.join(settings.reports_dir, f"weather_report_{day_yyyy_mm_dd}.md")

    if not stats:
        content = (
            f"# Daily Weather Report\n\n"
            f"- **Date:** {day_yyyy_mm_dd}\n"
            f"- **Location:** {settings.location_name}\n\n"
            f"✅ No data collected for this date yet.\n"
        )
        with open(filename, "w", encoding="utf-8") as f:
            f.write(content)
        return filename

    content = []
    content.append("# Daily Weather Report\n")
    content.append(f"- **Date:** {day_yyyy_mm_dd}")
    content.append(f"- **Location:** {settings.location_name}")
    content.append(f"- **Data points collected:** {stats['count']}\n")

    content.append("## Summary\n")
    content.append(f"- **Temperature (°C):** min {format_num(stats['temp_min'])}, "
                   f"avg {format_num(stats['temp_avg'])}, max {format_num(stats['temp_max'])}")
    content.append(f"- **Humidity (%):** min {format_num(stats['hum_min'])}, "
                   f"avg {format_num(stats['hum_avg'])}, max {format_num(stats['hum_max'])}")
    content.append(f"- **Max wind (km/h):** {format_num(stats['wind_max'])}")
    content.append(f"- **Total rain (mm):** {format_num(stats['rain_total'])}\n")

    # A small “rule-based note” (still NOT an agent)
    notes = []
    if stats["temp_max"] >= 35:
        notes.append("🔥 It was a hot day. Drink water and avoid direct sun in afternoon.")
    if stats["rain_total"] >= 10:
        notes.append("🌧️ Significant rain. Roads may be slippery.")
    if stats["wind_max"] >= 35:
        notes.append("💨 Strong winds occurred. Be careful outdoors.")

    content.append("## Notes (Rule-Based)\n")
    if notes:
        for n in notes:
            content.append(f"- {n}")
    else:
        content.append("- No special conditions detected.\n")

    if latest:
        content.append("## Latest Reading\n")
        content.append(f"- **Time:** {latest['observed_at']}")
        content.append(f"- **Temp:** {format_num(latest['temperature_c'])} °C")
        content.append(f"- **Humidity:** {format_num(latest['humidity_percent'])} %")
        content.append(f"- **Wind:** {format_num(latest['wind_kph'])} km/h")
        content.append(f"- **Rain:** {format_num(latest['rain_mm'])} mm\n")

    with open(filename, "w", encoding="utf-8") as f:
        f.write("\n".join(content))

    return filename

def today_str() -> str:
    return datetime.now().strftime("%Y-%m-%d")

def yesterday_str() -> str:
    return (datetime.now() - timedelta(days=1)).strftime("%Y-%m-%d")
```

## 8) app.py (Run workflow)

```python
from config import get_settings
from collect import run_collection
from report import generate_daily_report, today_str

def main():
    settings = get_settings()

    # Step 1: Collect data
    row = run_collection(settings)
    print("✅ Collected & stored:", row)

    # Step 2: Generate report (for today)
    day = today_str()
    report_file = generate_daily_report(settings, day)
    print("📄 Report generated:", report_file)

if __name__ == "__main__":
    main()
```

```md
# Project 3: Data Collection & Reporting Workflow (No Agent)

## What it does
1) Collects current weather data (Open-Meteo, free, no API key)
2) Stores it in SQLite database
3) Generates a daily Markdown report in /reports

## Setup
pip install -r requirements.txt

Create .env from .env.example and edit values.

## Run
python app.py

## Run it multiple times in a day
If you run it many times, you will collect multiple data points.
The daily report will summarize all points collected today.

## Scheduling (optional)
- Windows: use Task Scheduler to run `python app.py` every hour
- Linux/Mac: cron job every hour
```
