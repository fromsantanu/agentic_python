# Project 9 
## Monitoring & Follow-Up Agent

This project teaches a **very realistic and powerful pattern**:

> **Do nothing most of the time.
> Watch continuously.
> Act only when something changes or crosses a limit.**

This is how:

* Patient follow-up systems work
* ICU monitors work
* Server monitoring works
* Chronic disease management works

---

## Core idea (very simple)

Think of a nurse monitoring a patient:

* Vital signs are checked regularly
* If everything is normal → no action
* If something worsens → doctor is informed
* If it improves → monitoring continues

👉 That’s exactly what this agent does.

---

# What this system does

1. **Workflow**

   * Periodically collects data
   * Stores historical data
   * Compares current vs past

2. **Agent**

   * Called **only if change is detected**
   * Decides:

     * Follow-up needed?
     * Escalate?
     * Continue monitoring?

# 3. monitor/collector.py

*(Routine monitoring – no intelligence)*

```python
import requests

def collect_weather(latitude: float, longitude: float, timezone: str) -> dict:
    url = (
        "https://api.open-meteo.com/v1/forecast"
        f"?latitude={latitude}&longitude={longitude}"
        "&current=temperature_2m,relative_humidity_2m,wind_speed_10m"
        f"&timezone={timezone}"
    )

    response = requests.get(url, timeout=20)
    response.raise_for_status()
    data = response.json()["current"]

    return {
        "temperature": data["temperature_2m"],
        "humidity": data["relative_humidity_2m"],
        "wind": data["wind_speed_10m"],
        "time": data["time"]
    }
```

# 4. monitor/storage.py

*(Store historical data)*

```python
import json
import os

DATA_PATH = "data/history.json"

def load_history() -> list:
    if not os.path.exists(DATA_PATH):
        return []
    with open(DATA_PATH, "r") as f:
        return json.load(f)

def save_history(history: list):
    os.makedirs("data", exist_ok=True)
    with open(DATA_PATH, "w") as f:
        json.dump(history, f, indent=2)
```

# 5. monitor/comparator.py

*(Detect change – still no AI)*

```python
def detect_change(previous: dict, current: dict) -> dict:
    """
    Simple rule-based comparison
    """
    changes = {}

    if abs(current["temperature"] - previous["temperature"]) >= 5:
        changes["temperature"] = {
            "from": previous["temperature"],
            "to": current["temperature"]
        }

    if abs(current["humidity"] - previous["humidity"]) >= 15:
        changes["humidity"] = {
            "from": previous["humidity"],
            "to": current["humidity"]
        }

    if abs(current["wind"] - previous["wind"]) >= 20:
        changes["wind"] = {
            "from": previous["wind"],
            "to": current["wind"]
        }

    return {
        "changed": len(changes) > 0,
        "details": changes
    }
```

# 6. agent/schemas.py

*(Structured follow-up decision)*

from pydantic import BaseModel
from typing import Literal

class FollowUpInput(BaseModel):
    previous: dict
    current: dict
    changes: dict

class FollowUpDecision(BaseModel):
    action: Literal["IGNORE", "FOLLOW_UP", "ESCALATE"]
    reason: str
    message: str
    confidence: float

    response = client.responses.create(
        model=model,
        instructions=system_prompt,
        input=user_prompt,

# 8. agent/followup_agent.py

*(Agent decides what to do about the change)*

```python
import json
from pydantic import ValidationError
from agent.schemas import FollowUpInput, FollowUpDecision
from agent.llm import ask_llm

SYSTEM_PROMPT = """
You are a monitoring follow-up agent.

You receive:
- Previous data
- Current data
- Detected changes

Decide:
- IGNORE → no action needed
- FOLLOW_UP → continue monitoring / notify lightly
- ESCALATE → urgent attention needed

Be conservative and safety-focused.

Return ONLY valid JSON:
{
  "action": "IGNORE|FOLLOW_UP|ESCALATE",
  "reason": "...",
  "message": "...",
  "confidence": 0.0-1.0
}
"""

def evaluate(model: str, data: FollowUpInput) -> FollowUpDecision:
    payload = json.dumps(data.model_dump(), indent=2)
    raw = ask_llm(model, SYSTEM_PROMPT, payload)

    try:
        return FollowUpDecision.model_validate_json(raw)
    except ValidationError:
        raw2 = ask_llm(model, SYSTEM_PROMPT + "\nReturn JSON only.", payload)
        return FollowUpDecision.model_validate_json(raw2)
```

# 10. app.py

*(Main monitoring loop)*

```python
import os
from dotenv import load_dotenv

from monitor.collector import collect_weather
from monitor.storage import load_history, save_history
from monitor.comparator import detect_change
from agent.schemas import FollowUpInput
from agent.followup_agent import evaluate
from tools.notifier import notify

load_dotenv()

def env(name, default=""):
    return os.getenv(name, default)

def main():
    model = env("MODEL_NAME", "gpt-5.2")

    current = collect_weather(
        float(env("LATITUDE")),
        float(env("LONGITUDE")),
        env("TIMEZONE")
    )

    history = load_history()

    # First run → just store
    if not history:
        history.append(current)
        save_history(history)
        print("Initial data stored. Monitoring started.")
        return

    previous = history[-1]
    comparison = detect_change(previous, current)

    history.append(current)
    save_history(history)

    if not comparison["changed"]:
        print("No significant change detected. Continuing monitoring.")
        return

    # 🚨 Change detected → call agent
    followup_input = FollowUpInput(
        previous=previous,
        current=current,
        changes=comparison["details"]
    )

    decision = evaluate(model, followup_input)

    notify(decision.action, decision.message)

    print("\nDecision reason:", decision.reason)
    print("Confidence:", decision.confidence)

if __name__ == "__main__":
    main()
```
