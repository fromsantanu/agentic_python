## Manager–Worker Agent System

This is a **major milestone project**.
It looks very close to how **real organizations and real AI systems work**.

---

## Core idea (very simple)

Think of a company:

* **Manager**

  * Understands the goal
  * Breaks work into tasks
  * Assigns tasks to specialists

* **Workers**

  * Do one specific job
  * Report back results

👉 The manager does **thinking & coordination**
👉 Workers do **execution**

That is exactly what we build.

---

## What this system does

1. **Manager Agent**

   * Receives a goal
   * Breaks it into tasks
   * Assigns tasks to worker agents

2. **Worker Agents**

   * Each has a fixed role
   * Executes its assigned task
   * Returns results

3. **Manager**

   * Collects results
   * Produces final outcome

# 3. manager/schemas.py

*(How manager thinks and assigns work)*

```python
from pydantic import BaseModel
from typing import List, Literal

WorkerType = Literal["WEATHER", "REPORT", "SUMMARY"]

class TaskAssignment(BaseModel):
    task_id: str
    worker: WorkerType
    instruction: str

class ManagerPlan(BaseModel):
    goal: str
    tasks: List[TaskAssignment]

class ManagerResult(BaseModel):
    final_output: str
```

# 4. manager/llm.py

*(LLM wrapper)*

```python
from openai import OpenAI

client = OpenAI()

def ask_llm(model: str, system_prompt: str, user_prompt: str) -> str:
    response = client.responses.create(
        model=model,
        instructions=system_prompt,
        input=user_prompt,
    )
    return response.output_text
```

# 5. manager/manager_agent.py

*(Manager creates task plan)*

```python
import json
from pydantic import ValidationError
from manager.schemas import ManagerPlan
from manager.llm import ask_llm

SYSTEM_PROMPT = """
You are a manager agent.

Your job:
- Break the goal into small tasks
- Assign each task to ONE worker:
  WEATHER, REPORT, SUMMARY

Rules:
- WEATHER → collect weather data
- REPORT → create a report file
- SUMMARY → write a short human-friendly summary

Return ONLY valid JSON:
{
  "goal": "...",
  "tasks": [
    {"task_id":"1","worker":"WEATHER","instruction":"..."}
  ]
}
"""

def create_plan(model: str, goal: str) -> ManagerPlan:
    raw = ask_llm(model, SYSTEM_PROMPT, goal)
    try:
        return ManagerPlan.model_validate_json(raw)
    except ValidationError:
        raw2 = ask_llm(model, SYSTEM_PROMPT + "\nReturn JSON only.", goal)
        return ManagerPlan.model_validate_json(raw2)
```

# 6. workers/weather_worker.py

*(Worker 1 – data collection)*

```python
import requests

def run(latitude: float, longitude: float, timezone: str) -> dict:
    url = (
        "https://api.open-meteo.com/v1/forecast"
        f"?latitude={latitude}&longitude={longitude}"
        "&current=temperature_2m,relative_humidity_2m,wind_speed_10m"
        f"&timezone={timezone}"
    )

    r = requests.get(url, timeout=20)
    r.raise_for_status()
    data = r.json()["current"]

    return {
        "temperature": data["temperature_2m"],
        "humidity": data["relative_humidity_2m"],
        "wind": data["wind_speed_10m"],
        "time": data["time"]
    }
```

# 7. workers/report_worker.py

*(Worker 2 – report writer)*

```python
import os
from datetime import datetime

def run(weather: dict, location: str) -> str:
    os.makedirs("reports", exist_ok=True)
    date = datetime.now().strftime("%Y-%m-%d")
    path = f"reports/weather_{date}.txt"

    with open(path, "w") as f:
        f.write(f"Weather Report – {location}\n")
        f.write("--------------------------------\n")
        for k, v in weather.items():
            f.write(f"{k}: {v}\n")

    return path
```

# 9. app.py

*(Orchestration – manager assigns, workers execute)*

```python
import os
import json
from datetime import datetime
from dotenv import load_dotenv

from manager.manager_agent import create_plan
from workers.weather_worker import run as weather_worker
from workers.report_worker import run as report_worker
from workers.summary_worker import run as summary_worker

load_dotenv()

def env(name, default=""):
    return os.getenv(name, default)

def log(record: dict):
    os.makedirs("data", exist_ok=True)
    with open("data/execution_log.jsonl", "a") as f:
        f.write(json.dumps(record) + "\n")

def main():
    model = env("MODEL_NAME", "gpt-5.2")

    goal = "Prepare today's weather report with a short summary."

    plan = create_plan(model, goal)

    print("\n=== MANAGER PLAN ===")
    for t in plan.tasks:
        print(t.task_id, t.worker, t.instruction)

    results = {}
    weather_data = None
    report_path = None
    summary_text = None

    for task in plan.tasks:
        if task.worker == "WEATHER":
            weather_data = weather_worker(
                float(env("LATITUDE")),
                float(env("LONGITUDE")),
                env("TIMEZONE")
            )
            results[task.task_id] = weather_data

        elif task.worker == "REPORT":
            report_path = report_worker(
                weather_data,
                env("LOCATION_NAME")
            )
            results[task.task_id] = report_path

        elif task.worker == "SUMMARY":
            summary_text = summary_worker(
                weather_data,
                env("LOCATION_NAME")
            )
            results[task.task_id] = summary_text

    print("\n=== FINAL OUTPUT ===")
    print("Report:", report_path)
    print("Summary:", summary_text)

    log({
        "time": datetime.now().isoformat(),
        "goal": goal,
        "results": results
    })

if __name__ == "__main__":
    main()
```
