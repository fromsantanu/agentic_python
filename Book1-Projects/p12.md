## Big picture (very simple words)

Think of a **real hospital system**:

1. **Intake agent**
   → collects patient information

2. **Decision agent**
   → decides what should be done

3. **Action agent**
   → performs the task (report, alert, escalation)

4. **Monitoring agent**
   → watches what happens next and follows up

Each agent has **one clear responsibility**.
Together, they form a **complete intelligent system**.

* Workflow + agents working together

# 4. decision/schemas.py

```python
from pydantic import BaseModel
from typing import Literal

DecisionType = Literal["NORMAL", "WARNING", "CRITICAL"]

class DecisionInput(BaseModel):
    intake: dict
    observation: dict

class DecisionOutput(BaseModel):
    level: DecisionType
    reason: str
    recommended_action: str
```

---

# 5. decision/decision_agent.py

*(Thinks and decides severity)*

```python
import json
from pydantic import ValidationError
from openai import OpenAI
from decision.schemas import DecisionInput, DecisionOutput

client = OpenAI()

SYSTEM_PROMPT = """
You are a decision agent.

Based on the observation and intake:
- Decide severity: NORMAL, WARNING, CRITICAL
- Be conservative and safety-focused

Return JSON only:
{
  "level": "...",
  "reason": "...",
  "recommended_action": "..."
}
"""

def decide(model: str, data: DecisionInput) -> DecisionOutput:
    raw = client.responses.create(
        model=model,
        instructions=SYSTEM_PROMPT,
        input=json.dumps(data.model_dump(), indent=2)
    ).output_text

    try:
        return DecisionOutput.model_validate_json(raw)
    except ValidationError:
        raw2 = client.responses.create(
            model=model,
            instructions=SYSTEM_PROMPT + "\nReturn JSON only.",
            input=json.dumps(data.model_dump(), indent=2)
        ).output_text
        return DecisionOutput.model_validate_json(raw2)
```

# 6. action/action_agent.py

*(Executes based on decision)*

```python
import os
from datetime import datetime

def act(decision: dict, observation: dict, location: str) -> dict:
    os.makedirs("reports", exist_ok=True)

    date = datetime.now().strftime("%Y-%m-%d")
    report_path = f"reports/system_report_{date}.txt"

    with open(report_path, "w") as f:
        f.write(f"Location: {location}\n")
        f.write(f"Decision Level: {decision['level']}\n")
        f.write(f"Reason: {decision['reason']}\n\n")
        f.write("Observation:\n")
        for k, v in observation.items():
            f.write(f"{k}: {v}\n")

    return {
        "report": report_path,
        "status": "action_completed"
    }
```

# 7. monitoring/monitoring_agent.py

*(Watches outcomes and decides follow-up)*

```python
def monitor(decision: dict) -> dict:
    if decision["level"] == "CRITICAL":
        return {
            "follow_up": "Immediate human intervention required",
            "monitoring": "continuous"
        }

    if decision["level"] == "WARNING":
        return {
            "follow_up": "Recheck after few hours",
            "monitoring": "periodic"
        }

    return {
        "follow_up": "No action needed",
        "monitoring": "routine"
    }
```

# 8. tools/weather.py

*(Data source)*

```python
import requests

def get_weather(lat: float, lon: float, tz: str) -> dict:
    url = (
        "https://api.open-meteo.com/v1/forecast"
        f"?latitude={lat}&longitude={lon}"
        "&current=temperature_2m,relative_humidity_2m,wind_speed_10m"
        f"&timezone={tz}"
    )

    r = requests.get(url, timeout=20)
    r.raise_for_status()
    cur = r.json()["current"]

    return {
        "temperature": cur["temperature_2m"],
        "humidity": cur["relative_humidity_2m"],
        "wind": cur["wind_speed_10m"],
        "time": cur["time"]
    }
```

# 10. app.py

*(End-to-end orchestration)*

```python
import os
import json
from datetime import datetime
from dotenv import load_dotenv

from intake.intake_agent import collect_input
from tools.weather import get_weather
from decision.schemas import DecisionInput
from decision.decision_agent import decide
from action.action_agent import act
from monitoring.monitoring_agent import monitor
from tools.notifier import notify

load_dotenv()

def env(name, default=""):
    return os.getenv(name, default)

def log(record: dict):
    os.makedirs("data", exist_ok=True)
    with open("data/system_log.jsonl", "a") as f:
        f.write(json.dumps(record) + "\n")

def main():
    model = env("MODEL_NAME", "gpt-5.2")

    # 1. Intake
    intake_data = collect_input()

    # 2. Observation
    observation = get_weather(
        float(env("LATITUDE")),
        float(env("LONGITUDE")),
        env("TIMEZONE")
    )

    # 3. Decision
    decision_input = DecisionInput(
        intake=intake_data,
        observation=observation
    )

    decision = decide(model, decision_input)

    # 4. Action
    action_result = act(
        decision.model_dump(),
        observation,
        env("LOCATION_NAME")
    )

    # 5. Monitoring
    follow_up = monitor(decision.model_dump())

    # Notify human (optional)
    if decision.level != "NORMAL":
        notify(f"{decision.level}: {decision.reason}")

    # Output
    print("\n=== END-TO-END RESULT ===")
    print("Decision :", decision.level)
    print("Action   :", action_result)
    print("Monitor  :", follow_up)

    log({
        "time": datetime.now().isoformat(),
        "intake": intake_data,
        "observation": observation,
        "decision": decision.model_dump(),
        "action": action_result,
        "monitoring": follow_up
    })

if __name__ == "__main__":
    main()
```
