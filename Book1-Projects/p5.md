Below is a **complete, runnable codebase** for **Project 5: Goal-Oriented Agent**.

This agent can take a goal like:

> “Create today’s weather report and email it to me.”

…and it will:

1. **Break the goal into steps** (plan)
2. **Execute steps one by one** (act)
3. **Stop when done** (finish)
4. **Save progress** so it can **resume** if you run it again

We’ll use OpenAI’s **Responses API** for the “brain”. ([OpenAI Platform][1])
And we’ll store the API key in environment variables, as OpenAI recommends. ([OpenAI Platform][2])

```python
from pydantic import BaseModel, Field
from typing import List, Literal, Optional, Dict, Any

# A plan step is a single, small action.
class PlanStep(BaseModel):
    id: str
    title: str
    tool_name: str
    args: Dict[str, Any] = Field(default_factory=dict)

class Plan(BaseModel):
    goal: str
    steps: List[PlanStep]
    success_criteria: List[str]

# The agent can either call a tool or finish.
ActionType = Literal["CALL_TOOL", "FINISH"]

class AgentAction(BaseModel):
    type: ActionType
    tool_name: Optional[str] = None
    args: Dict[str, Any] = Field(default_factory=dict)
    reason: str
```

```python
client = OpenAI()

def ask_model(model_name: str, instructions: str, user_text: str) -> str:
    """
    Minimal wrapper around Responses API.
    """
    resp = client.responses.create(
        model=model_name,
        instructions=instructions,
        input=user_text,
    )
```

# agent/memory.py

```python
import json
import os
from typing import Dict, Any

def ensure_parent(path: str) -> None:
    parent = os.path.dirname(path)
    if parent:
        os.makedirs(parent, exist_ok=True)

def load_state(path: str) -> Dict[str, Any]:
    if not os.path.exists(path):
        return {
            "goal": None,
            "plan": None,
            "step_index": 0,
            "tool_results": [],
            "done": False,
        }
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

def save_state(path: str, state: Dict[str, Any]) -> None:
    ensure_parent(path)
    with open(path, "w", encoding="utf-8") as f:
        json.dump(state, f, indent=2, ensure_ascii=False)
```

# tools/registry.py

```python
from typing import Callable, Dict, Any

ToolFn = Callable[[Dict[str, Any]], Dict[str, Any]]

class ToolRegistry:
    def __init__(self) -> None:
        self.tools: Dict[str, ToolFn] = {}

    def register(self, name: str, fn: ToolFn) -> None:
        self.tools[name] = fn

    def list_tools(self) -> list[str]:
        return sorted(self.tools.keys())

    def run(self, name: str, args: Dict[str, Any]) -> Dict[str, Any]:
        if name not in self.tools:
            return {"ok": False, "error": f"Unknown tool: {name}"}
        try:
            return self.tools[name](args)
        except Exception as e:
            return {"ok": False, "error": str(e)}
```

Free Open-Meteo API (no key). Same idea as your earlier projects.

```python
import requests
from typing import Dict, Any

def fetch_weather(args: Dict[str, Any]) -> Dict[str, Any]:
    lat = float(args["latitude"])
    lon = float(args["longitude"])
    tz = str(args.get("timezone", "UTC"))

    url = (
        "https://api.open-meteo.com/v1/forecast"
        f"?latitude={lat}&longitude={lon}"
        "&current=temperature_2m,relative_humidity_2m,wind_speed_10m,precipitation"
        f"&timezone={tz}"
    )

    r = requests.get(url, timeout=20)
    r.raise_for_status()
    data = r.json()
    cur = data.get("current", {})

    return {
        "ok": True,
        "observed_at": str(cur.get("time")),
        "temperature_c": float(cur.get("temperature_2m", 0.0)),
        "humidity_percent": float(cur.get("relative_humidity_2m", 0.0)),
        "wind_kph": float(cur.get("wind_speed_10m", 0.0)),
        "rain_mm": float(cur.get("precipitation", 0.0)),
    }
```

```python
import os
from datetime import datetime
from typing import Dict, Any

def write_weather_report(args: Dict[str, Any]) -> Dict[str, Any]:
    reports_dir = args["reports_dir"]
    location_name = args.get("location_name", "Unknown")

    weather = args["weather"]  # dict with temperature, etc.

    os.makedirs(reports_dir, exist_ok=True)
    day = datetime.now().strftime("%Y-%m-%d")
    filename = os.path.join(reports_dir, f"weather_report_{day}.md")

    content = []
    content.append("# Daily Weather Report\n")
    content.append(f"- **Date:** {day}")
    content.append(f"- **Location:** {location_name}\n")

    content.append("## Current Reading\n")
    content.append(f"- **Observed at:** {weather.get('observed_at')}")
    content.append(f"- **Temperature:** {weather.get('temperature_c')} °C")
    content.append(f"- **Humidity:** {weather.get('humidity_percent')} %")
    content.append(f"- **Wind:** {weather.get('wind_kph')} km/h")
    content.append(f"- **Rain:** {weather.get('rain_mm')} mm\n")

    with open(filename, "w", encoding="utf-8") as f:
        f.write("\n".join(content))

    return {"ok": True, "report_file": filename}
```

# tools/emailer.py  (Tool: send_email)

```python
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from typing import Dict, Any

def send_email(args: Dict[str, Any]) -> Dict[str, Any]:
    smtp_host = args["smtp_host"]
    smtp_port = int(args["smtp_port"])
    username = args["smtp_username"]
    password = args["smtp_password"]
    to_email = args["to_email"]

    subject = args["subject"]
    body_text = args["body_text"]

    msg = MIMEMultipart()
    msg["From"] = username
    msg["To"] = to_email
    msg["Subject"] = subject
    msg.attach(MIMEText(body_text, "plain", "utf-8"))

    with smtplib.SMTP(smtp_host, smtp_port) as server:
        server.starttls()
        server.login(username, password)
        server.send_message(msg)

    return {"ok": True}
```

# agent/planner.py  (Goal → Plan)

This is the “goal-oriented” part: it turns a goal into steps.

```python
import json
from pydantic import ValidationError
from agent.schemas import Plan
from agent.llm import ask_model

PLANNER_INSTRUCTIONS = """
You are a planner agent.
Create a short plan that reaches the goal using the available tools.

Rules:
- Use ONLY these tool names: fetch_weather, write_weather_report, send_email
- Each step must be small and executable
- Output JSON ONLY that matches this schema:

{
  "goal": "...",
  "steps": [
    {"id":"step1","title":"...","tool_name":"...","args":{...}}
  ],
  "success_criteria": ["..."]
}
"""

def create_plan(model_name: str, goal: str, tool_info_text: str, defaults: dict) -> Plan:
    user_text = json.dumps({
        "goal": goal,
        "available_tools": tool_info_text,
        "defaults": defaults
    }, indent=2)

    raw = ask_model(model_name, PLANNER_INSTRUCTIONS, user_text)

    # validate
    try:
        return Plan.model_validate_json(raw)
    except ValidationError as e:
        # One simple retry: ask model to output valid JSON only.
        repair_instructions = PLANNER_INSTRUCTIONS + "\nReturn VALID JSON only. No extra text."
        raw2 = ask_model(model_name, repair_instructions, user_text)
        return Plan.model_validate_json(raw2)
```

# agent/runner.py  (Execute steps one by one)

```python
from typing import Dict, Any
from agent.schemas import Plan
from tools.registry import ToolRegistry

def run_next_step(plan: Plan, state: Dict[str, Any], registry: ToolRegistry) -> Dict[str, Any]:
    idx = int(state.get("step_index", 0))
    if idx >= len(plan.steps):
        state["done"] = True
        return {"ok": True, "message": "All steps completed."}

    step = plan.steps[idx]

    # Merge step args with runtime context (e.g., carry weather result to report step)
    args = dict(step.args)

    # If the step expects "weather" but planner didn’t fill it, we try to fetch it from previous tool results.
    if step.tool_name == "write_weather_report":
        # find the last fetch_weather result
        last_weather = None
        for r in reversed(state.get("tool_results", [])):
            if r.get("tool_name") == "fetch_weather" and r.get("result", {}).get("ok"):
                last_weather = r["result"]
                break
        if last_weather:
            args["weather"] = last_weather

    result = registry.run(step.tool_name, args)

    state.setdefault("tool_results", [])
    state["tool_results"].append({
        "step_id": step.id,
        "tool_name": step.tool_name,
        "args": args,
        "result": result
    })

    # Move forward only if ok
    if result.get("ok"):
        state["step_index"] = idx + 1
    else:
        # stop here; user can fix config and re-run
        state["done"] = False

    return result
```

# app.py  (Main entry point)

```python
import os
from dotenv import load_dotenv

from agent.memory import load_state, save_state
from agent.planner import create_plan
from agent.runner import run_next_step
from tools import build_registry

load_dotenv()

def env(name: str, default: str = "") -> str:
    return os.getenv(name, default).strip()

def main():
    model_name = env("MODEL_NAME", "gpt-5.2")
    memory_path = env("MEMORY_PATH", "data/agent_state.json")

    # Defaults the planner can reuse
    defaults = {
        "latitude": float(env("LATITUDE", "0")),
        "longitude": float(env("LONGITUDE", "0")),
        "timezone": env("TIMEZONE", "UTC"),
        "location_name": env("LOCATION_NAME", "Unknown"),
        "reports_dir": env("REPORTS_DIR", "reports"),
        "smtp_host": env("SMTP_HOST", ""),
        "smtp_port": int(env("SMTP_PORT", "587") or "587"),
        "smtp_username": env("SMTP_USERNAME", ""),
        "smtp_password": env("SMTP_PASSWORD", ""),
        "to_email": env("ALERT_TO_EMAIL", ""),
    }

    registry = build_registry()

    # Load state (resume support)
    state = load_state(memory_path)

    if not state.get("goal"):
        # First run: set the goal here (change this line for your demo)
        goal = "Fetch current weather, generate today's weather report, and email the report summary."
        state["goal"] = goal

    # If no plan yet, create plan
    if not state.get("plan"):
        tool_info = (
            "fetch_weather(args: latitude, longitude, timezone)\n"
            "write_weather_report(args: reports_dir, location_name, weather)\n"
            "send_email(args: smtp_host, smtp_port, smtp_username, smtp_password, to_email, subject, body_text)\n"
        )

        # Give the planner sensible default arguments
        plan = create_plan(
            model_name=model_name,
            goal=state["goal"],
            tool_info_text=tool_info,
            defaults=defaults,
        )

        # Fill missing args using defaults (simple merge)
        for step in plan.steps:
            if step.tool_name == "fetch_weather":
                step.args.setdefault("latitude", defaults["latitude"])
                step.args.setdefault("longitude", defaults["longitude"])
                step.args.setdefault("timezone", defaults["timezone"])

            if step.tool_name == "write_weather_report":
                step.args.setdefault("reports_dir", defaults["reports_dir"])
                step.args.setdefault("location_name", defaults["location_name"])

            if step.tool_name == "send_email":
                step.args.setdefault("smtp_host", defaults["smtp_host"])
                step.args.setdefault("smtp_port", defaults["smtp_port"])
                step.args.setdefault("smtp_username", defaults["smtp_username"])
                step.args.setdefault("smtp_password", defaults["smtp_password"])
                step.args.setdefault("to_email", defaults["to_email"])
                step.args.setdefault("subject", "Daily Weather Report")
                step.args.setdefault("body_text", "Report created. Please check the attached folder or log output.")

        state["plan"] = plan.model_dump()
        state["step_index"] = 0
        state["tool_results"] = []
        state["done"] = False
        save_state(memory_path, state)

        print("\n✅ Plan created:")
        for s in plan.steps:
            print(f"- {s.id}: {s.title}  (tool: {s.tool_name})")

    # Execute steps until a failure or done
    from agent.schemas import Plan
    plan = Plan(**state["plan"])

    while not state.get("done", False):
        result = run_next_step(plan, state, registry)
        save_state(memory_path, state)

        if not result.get("ok"):
            print("\n❌ Step failed:")
            print(result)
            print("\nFix the issue (usually email config) and run again. Agent will resume.")
            return

        print("\n✅ Step result:")
        print(result)

        # If finished all steps
        if state.get("step_index", 0) >= len(plan.steps):
            state["done"] = True
            save_state(memory_path, state)

    print("\n🎉 Goal completed!")
    print(f"State saved at: {memory_path}")

if __name__ == "__main__":
    main()
```

# README.md

```md
# Project 5: Goal-Oriented Agent

## What it does
- Takes a goal
- Plans steps
- Executes step-by-step using tools
- Saves state so it can resume

## Why it's a "goal-oriented agent"
Because it doesn't just do one fixed workflow.
It decides the steps needed to reach the goal.

## Setup
pip install -r requirements.txt

Create `.env` from `.env.example` and set:
OPENAI_API_KEY=...

## Run
python app.py

## Resume behavior
If a step fails (often email settings), fix `.env` and run again.
The agent resumes from the same step using `data/agent_state.json`.
```

## How this is different from Project 4 (simple words)

* **Project 4**: “Pick one action (do now / schedule / ignore)”
* **Project 5**: “Make a plan and **finish the whole goal**”

---

## Common mistakes

1. **No `OPENAI_API_KEY` in `.env`** (agent can’t think) ([OpenAI Platform][2])
2. Email step fails because Gmail needs **App Password**
3. Wrong latitude/longitude → wrong city weather
4. Expecting the agent to “magically know tools” (we must list tools clearly)
