# Project 2: 
## Rule-Based Email / Alert System (workflow without agent).

### How does it work

* You define **rules** (example: “If temperature > 35°C → send email alert”)
* The program **checks the data**
* If rule matches → it **sends an email**
* No AI, no thinking. Only fixed logic.

## 3) config.py

```python
import os
from dataclasses import dataclass
from dotenv import load_dotenv

load_dotenv()

@dataclass(frozen=True)
class Settings:
    smtp_host: str = os.getenv("SMTP_HOST", "")
    smtp_port: int = int(os.getenv("SMTP_PORT", "587"))
    smtp_username: str = os.getenv("SMTP_USERNAME", "")
    smtp_password: str = os.getenv("SMTP_PASSWORD", "")
    alert_to_email: str = os.getenv("ALERT_TO_EMAIL", "")

    latitude: float = float(os.getenv("LATITUDE", "0"))
    longitude: float = float(os.getenv("LONGITUDE", "0"))
    timezone: str = os.getenv("TIMEZONE", "UTC")

    system_name: str = os.getenv("ALERT_SYSTEM_NAME", "RuleBasedAlerts")

def get_settings() -> Settings:
    s = Settings()
    missing = []
    if not s.smtp_host: missing.append("SMTP_HOST")
    if not s.smtp_username: missing.append("SMTP_USERNAME")
    if not s.smtp_password: missing.append("SMTP_PASSWORD")
    if not s.alert_to_email: missing.append("ALERT_TO_EMAIL")

    if missing:
        raise ValueError(f"Missing environment variables: {', '.join(missing)}. "
                         f"Create a .env file based on .env.example")
    return s
```

## 4) emailer.py

```python
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart

def send_email(
    smtp_host: str,
    smtp_port: int,
    username: str,
    password: str,
    to_email: str,
    subject: str,
    body_text: str,
) -> None:
    msg = MIMEMultipart()
    msg["From"] = username
    msg["To"] = to_email
    msg["Subject"] = subject

    msg.attach(MIMEText(body_text, "plain", "utf-8"))

    with smtplib.SMTP(smtp_host, smtp_port) as server:
        server.starttls()
        server.login(username, password)
        server.send_message(msg)
```

## 5) providers/weather_openmeteo.py

This uses **Open-Meteo** (free, no API key) to fetch current weather.

```python
import requests

def get_current_weather(latitude: float, longitude: float, timezone: str) -> dict:
    """
    Returns a dictionary with:
      temperature_c, wind_kph, humidity_percent, rain_mm
    """
    url = (
        "https://api.open-meteo.com/v1/forecast"
        f"?latitude={latitude}&longitude={longitude}"
        "&current=temperature_2m,relative_humidity_2m,wind_speed_10m,precipitation"
        f"&timezone={timezone}"
    )

    r = requests.get(url, timeout=20)
    r.raise_for_status()
    data = r.json()

    current = data.get("current", {})
    return {
        "temperature_c": float(current.get("temperature_2m", 0.0)),
        "humidity_percent": float(current.get("relative_humidity_2m", 0.0)),
        "wind_kph": float(current.get("wind_speed_10m", 0.0)),
        "rain_mm": float(current.get("precipitation", 0.0)),
    }
```

## 7) rules.py

This is the “IF this THEN that” engine.

```python
import json
from typing import Callable, Any

OPS: dict[str, Callable[[Any, Any], bool]] = {
    ">": lambda a, b: a > b,
    ">=": lambda a, b: a >= b,
    "<": lambda a, b: a < b,
    "<=": lambda a, b: a <= b,
    "==": lambda a, b: a == b,
    "!=": lambda a, b: a != b,
}

def load_rules(path: str) -> list[dict]:
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

def render_template(text: str, data: dict) -> str:
    """
    Very simple template:
    "Temp is {{temperature_c}}" -> replaces with data value.
    """
    out = text
    for k, v in data.items():
        out = out.replace("{{" + k + "}}", str(v))
    return out

def rule_matches(rule: dict, data: dict) -> bool:
    cond = rule.get("when", {})
    field = cond.get("field")
    op = cond.get("op")
    value = cond.get("value")

    if field not in data:
        return False
    if op not in OPS:
        raise ValueError(f"Unsupported operator: {op}")

    return OPS[op](data[field], value)

def evaluate_rules(rules: list[dict], data: dict) -> list[dict]:
    """
    Returns a list of rules that matched.
    """
    matched = []
    for rule in rules:
        if rule_matches(rule, data):
            matched.append(rule)
    return matched
```

## 8) app.py (Main runner)

```python
from datetime import datetime
from config import get_settings
from emailer import send_email
from rules import load_rules, evaluate_rules, render_template
from providers.weather_openmeteo import get_current_weather

RULES_PATH = "rules/rules.json"

def main():
    settings = get_settings()

    # 1) Get data (example: weather)
    data = get_current_weather(
        latitude=settings.latitude,
        longitude=settings.longitude,
        timezone=settings.timezone
    )

    # Add metadata
    data["timestamp"] = datetime.now().isoformat(timespec="seconds")
    data["location"] = f"{settings.latitude},{settings.longitude}"

    print("✅ Current data:", data)

    # 2) Load rules
    rules = load_rules(RULES_PATH)

    # 3) Evaluate rules
    matched = evaluate_rules(rules, data)

    if not matched:
        print("ℹ️ No rules matched. No email sent.")
        return

    # 4) Send alerts (one email per matched rule)
    for rule in matched:
        subject = f"[{settings.system_name}] " + rule["email"]["subject"]
        body = rule["email"]["body"]

        # fill placeholders like {{temperature_c}}
        subject = render_template(subject, data)
        body = render_template(body, data)

        body = (
            body
            + "\n\n---\n"
            + f"Rule: {rule.get('name')}\n"
            + f"Time: {data['timestamp']}\n"
            + f"Location: {data['location']}\n"
            + f"Data: {data}\n"
        )

        print(f"📧 Sending alert: {rule.get('name')} -> {settings.alert_to_email}")
        send_email(
            smtp_host=settings.smtp_host,
            smtp_port=settings.smtp_port,
            username=settings.smtp_username,
            password=settings.smtp_password,
            to_email=settings.alert_to_email,
            subject=subject,
            body_text=body,
        )

    print("✅ Done. All matched alerts sent.")

if __name__ == "__main__":
    main()
```

# Project 2: Rule-Based Email / Alert System (No Agent)

## What it does
- Fetches current weather data (Open-Meteo, no API key)
- Checks JSON rules (IF this THEN that)

python app.py

Update rules/rules.json
Example:
- temperature_c > 35
- rain_mm > 5
