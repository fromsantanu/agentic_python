This project shows a **very important real-world pattern**:

> **The workflow does all routine work.
> The agent is called only when something unusual happens.**

Think of it like this (simple analogy):

* A **hospital receptionist** does normal registration (workflow)
* A **doctor** is called only if the case looks serious (agent)

# Project 7 – Agent-Assisted Workflow

*(Workflow first, Agent only for exceptions)*

---

## What this project demonstrates

* A **fixed workflow** runs every time
* It follows rules and thresholds
* If everything is normal → workflow finishes
* If something looks **abnormal** → **agent is called**
* Agent decides **what to do next**

This is how most **production AI systems** are actually built.

## 3. workflow/collector.py

*(Routine data collection – no intelligence)*

```python
import requests

def collect_weather(latitude: float, longitude: float, timezone: str) -> dict:
    url = (
        "https://api.open-meteo.com/v1/forecast"
        f"?latitude={latitude}&longitude={longitude}"
        "&current=temperature_2m,relative_humidity_2m,wind_speed_10m,precipitation"
        f"&timezone={timezone}"
    )

    response = requests.get(url, timeout=20)
    response.raise_for_status()
    data = response.json()["current"]

    return {
        "temperature_c": data["temperature_2m"],
        "humidity": data["relative_humidity_2m"],
        "wind_kph": data["wind_speed_10m"],
        "rain_mm": data["precipitation"],
        "observed_at": data["time"],
    }
```

## 4. workflow/validator.py

*(Rule-based checks – still no agent)*

```python
def validate_weather(data: dict) -> dict:
    issues = []

    if data["temperature_c"] >= 38:
        issues.append("Very high temperature")

    if data["rain_mm"] >= 20:
        issues.append("Heavy rainfall")

    if data["wind_kph"] >= 40:
        issues.append("Strong winds")

    return {
        "ok": len(issues) == 0,
        "issues": issues
    }
```

## 5. workflow/reporter.py

*(Simple report generation)*

```python
import os
from datetime import datetime

def generate_report(weather: dict, location: str, reports_dir: str) -> str:
    os.makedirs(reports_dir, exist_ok=True)

    date = datetime.now().strftime("%Y-%m-%d")
    path = os.path.join(reports_dir, f"weather_{date}.txt")

    with open(path, "w") as f:
        f.write(f"Weather Report – {location}\n")
        f.write("---------------------------------\n")
        for k, v in weather.items():
            f.write(f"{k}: {v}\n")

    return path
```

## 6. agent/schemas.py

*(Structured thinking)*

```python
from pydantic import BaseModel
from typing import List, Literal

class ExceptionInput(BaseModel):
    weather: dict
    issues: List[str]

class ExceptionDecision(BaseModel):
    action: Literal["IGNORE", "WARN", "ESCALATE"]
    reason: str
    message: str
```

```python
client = OpenAI()

def ask_llm(model: str, system_prompt: str, user_prompt: str) -> str:
    response = client.responses.create(
        model=model,
        instructions=system_prompt,
        input=user_prompt,
    )
    return response.output_text
```

## 8. agent/exception_agent.py

*(Agent only handles exceptions)*

```python
import json
from pydantic import ValidationError
from agent.schemas import ExceptionInput, ExceptionDecision
from agent.llm import ask_llm

SYSTEM_PROMPT = """
You are an exception-handling agent.

You receive:
- Weather data
- Detected issues

Your task:
- Decide how serious the situation is
- Choose ONE action:
  IGNORE, WARN, or ESCALATE

Return ONLY valid JSON.
"""

def handle_exception(model: str, data: ExceptionInput) -> ExceptionDecision:
    user_prompt = json.dumps(data.model_dump(), indent=2)

    raw = ask_llm(model, SYSTEM_PROMPT, user_prompt)

    try:
        return ExceptionDecision.model_validate_json(raw)
    except ValidationError:
        raw2 = ask_llm(model, SYSTEM_PROMPT + "\nReturn JSON only.", user_prompt)
        return ExceptionDecision.model_validate_json(raw2)
```

## 10. app.py

*(Main orchestration – this is the key part)*

```python
import os
import json
from datetime import datetime
from dotenv import load_dotenv

from workflow.collector import collect_weather
from workflow.validator import validate_weather
from workflow.reporter import generate_report
from agent.schemas import ExceptionInput
from agent.exception_agent import handle_exception
from tools.notifier import notify

load_dotenv()

def env(name, default=""):
    return os.getenv(name, default)

def log_event(event: dict):
    os.makedirs("data", exist_ok=True)
    with open("data/logs.jsonl", "a") as f:
        f.write(json.dumps(event) + "\n")

def main():
    model = env("MODEL_NAME", "gpt-5.2")

    weather = collect_weather(
        float(env("LATITUDE")),
        float(env("LONGITUDE")),
        env("TIMEZONE"),
    )

    validation = validate_weather(weather)

    report_path = generate_report(
        weather,
        env("LOCATION_NAME"),
        env("REPORTS_DIR")
    )

    print("Report generated:", report_path)

    if validation["ok"]:
        print("✅ Workflow completed normally.")
        log_event({"time": datetime.now().isoformat(), "status": "normal"})
        return

    # 🚨 Exception detected → call agent
    print("⚠️ Exception detected, invoking agent...")

    agent_input = ExceptionInput(
        weather=weather,
        issues=validation["issues"]
    )

    decision = handle_exception(model, agent_input)

    notify(decision.message, decision.action)

    log_event({
        "time": datetime.now().isoformat(),
        "status": "exception",
        "issues": validation["issues"],
        "decision": decision.model_dump()
    })

if __name__ == "__main__":
    main()
```

Many people think:

> “Agent should do everything”

In reality:

> **Agents are expensive and slow**
> **Workflows are fast and reliable**

So we combine them.

This project teaches:

* When **not** to use AI
* When AI adds real value
