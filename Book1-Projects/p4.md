* Then your code **executes** that choice

## 3) models.py (data shapes)

```python
from pydantic import BaseModel, Field
from typing import Literal, List, Optional, Dict, Any

DecisionType = Literal["DO_NOW", "SCHEDULE", "DELEGATE", "IGNORE"]

class Task(BaseModel):
    title: str
    details: str = ""
    urgency: int = Field(ge=1, le=5, description="1=low, 5=high")
    importance: int = Field(ge=1, le=5, description="1=low, 5=high")
    due_in_hours: Optional[int] = Field(default=None, description="If known")
    estimated_minutes: int = Field(ge=1, le=600, default=30)

class Decision(BaseModel):
    decision: DecisionType
    reason: str
    next_step: str
    confidence: float = Field(ge=0, le=1, default=0.7)
    tags: List[str] = []
    # Optional structured “plan” if the decision is DO_NOW
    plan_steps: List[str] = []
    # For SCHEDULE
    suggested_time: Optional[str] = None  # e.g. "today 6pm"
    # For DELEGATE
    delegate_to: Optional[str] = None

class AgentInput(BaseModel):
    goal: str
    task: Task
    options: List[DecisionType] = ["DO_NOW", "SCHEDULE", "DELEGATE", "IGNORE"]
    context: Dict[str, Any] = {}
```

## 4) memory.py (very simple memory)

```python
import json
import os
from typing import Any, Dict

def load_memory(path: str) -> Dict[str, Any]:
    if not os.path.exists(path):
        return {"history": []}
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)

def save_memory(path: str, memory: Dict[str, Any]) -> None:
    with open(path, "w", encoding="utf-8") as f:
        json.dump(memory, f, indent=2, ensure_ascii=False)

def add_to_history(memory: Dict[str, Any], item: Dict[str, Any]) -> None:
    memory.setdefault("history", [])
    memory["history"].append(item)
    # keep last 20 items
    memory["history"] = memory["history"][-20:]
```

## 6) agent.py (the decision-making brain)

Uses:

* OpenAI **Responses API** ([OpenAI Platform][1])
* Forces **structured JSON** output ([OpenAI Platform][2])

```python
from openai import OpenAI
from pydantic import BaseModel
from typing import Type, Dict, Any

from models import AgentInput, Decision

client = OpenAI()

SYSTEM_INSTRUCTIONS = """
You are a simple decision-making agent.

Your job:
- Read the goal, task details, and options.
- Choose ONE decision from the allowed options.
- Give a short reason and a clear next step.
- Be practical and safe.
- Output MUST follow the JSON schema exactly.
"""

def decide(agent_input: AgentInput) -> Decision:
    # We ask for strict JSON output so parsing is safe.
    # (Structured Outputs concept) :contentReference[oaicite:5]{index=5}
    response = client.responses.create(
        model=agent_input.context.get("model_name", "gpt-5.2"),
        instructions=SYSTEM_INSTRUCTIONS,
        input=[
            {
                "role": "user",
                "content": [
                    {
                        "type": "input_text",
                        "text": agent_input.model_dump_json(indent=2),
                    }
                ],
            }
        ],
        # Using "text.format" with JSON schema style is a common approach in Responses API usage.
        # Exact knobs can evolve; keep this minimal and robust.
    )

    text = response.output_text
    # The model should output JSON only.
    return Decision.model_validate_json(text)
```

> Note: The official docs show the Responses API usage pattern with `client.responses.create(...)`. ([OpenAI Platform][1])

## 7) actions.py (execute the decision)

This is how your “workflow” reacts to the agent’s choice.

```python
from models import Decision, Task

def execute_decision(decision: Decision, task: Task) -> None:
    print("\n=== AGENT DECISION ===")
    print("Decision   :", decision.decision)
    print("Reason     :", decision.reason)
    print("Next step  :", decision.next_step)
    print("Confidence :", decision.confidence)

    if decision.plan_steps:
        print("\nPlan steps:")
        for i, step in enumerate(decision.plan_steps, 1):
            print(f"  {i}. {step}")

    if decision.decision == "DO_NOW":
        print("\n✅ Action: Start now.")
        # In real life, here you might open a ticket, run a script, etc.

    elif decision.decision == "SCHEDULE":
        print("\n📅 Action: Schedule it.")
        print("Suggested time:", decision.suggested_time or "not provided")

    elif decision.decision == "DELEGATE":
        print("\n👥 Action: Delegate it.")
        print("Delegate to:", decision.delegate_to or "not provided")

    elif decision.decision == "IGNORE":
        print("\n🧹 Action: Ignore for now (or drop it).")
```

## 9) app.py (run everything)

```python
import json
from utils import get_env
from models import AgentInput, Task
from agent import decide
from actions import execute_decision
from memory import load_memory, save_memory, add_to_history

def main():
    model_name = get_env("MODEL_NAME", "gpt-5.2")
    memory_path = get_env("MEMORY_PATH", "memory.json")

    memory = load_memory(memory_path)

    # Load sample tasks
    with open("samples/tasks.json", "r", encoding="utf-8") as f:
        tasks = json.load(f)

    for item in tasks:
        goal = item["goal"]
        task = Task(**item["task"])

        agent_input = AgentInput(
            goal=goal,
            task=task,
            context={
                "model_name": model_name,
                "recent_history": memory.get("history", []),
            },
        )

        print("\n==============================")
        print("Task:", task.title)
        print("==============================")

        decision = decide(agent_input)
        execute_decision(decision, task)

        add_to_history(memory, {
            "task": task.model_dump(),
            "decision": decision.model_dump(),
        })

    save_memory(memory_path, memory)
    print("\n✅ Finished. Memory updated:", memory_path)

if __name__ == "__main__":
    main()
```

# Why this is an “Agent” (simple words)

A workflow is like:

> “Do exactly these steps.”

An agent is like:

> “Here is the situation… choose what to do.”

This project is the **first step** into that.
